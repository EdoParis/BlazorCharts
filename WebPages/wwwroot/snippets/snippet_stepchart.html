<StepChart Model="@model" HidePoints="false"/>

@{
    private Cartesiangram model = new Cartesiangram("X1", "Y1");

    List<DataPoint> points1 = new();
    List<DataPoint> points2 = new();

    for (int i = 0; i < 10; i++)
    {
        points1.Add(new DataPoint(){ X = i, Y = i });
        points2.Add(new DataPoint(){ X = i, Y = 2 * i});
    }

    model.AddSerie("Line1", Color.Red, points1);
    model.AddSerie("Line2", Color.Blue, points2);
}
